\newtheorem{definition}{Definition}

\begin{document}
\begin{definition}
	We say that an integer \(m\) divides an integer \(n\) if \(m>0\) and the ratio \(n/m\) is an integer. We denote this by \(n\symbol{92} m\). 
\end{definition}

\begin{definition}
	We say that an integer \(n\) is a multiple of an integer \(m\) if \(n=mk\) for some integer \(k\).
\end{definition}
